cmake_minimum_required(VERSION 3.14)
project(Staj26)

set(CMAKE_CXX_STANDARD 17)

set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)

add_subdirectory(thirdparty/raylib)

add_executable(${PROJECT_NAME}
    main.cpp
    DroneSystem.cpp

    # Legacy Game Code (Archived in kullanılmayan/)
    kullanılmayan/Player.cpp
    kullanılmayan/Map.cpp
    kullanılmayan/CameraController.cpp
    kullanılmayan/Collectible.cpp
    kullanılmayan/Radar.cpp
    kullanılmayan/ParticleSystem.cpp
    kullanılmayan/AudioManager.cpp
    kullanılmayan/Enemy.cpp
    kullanılmayan/SaveSystem.cpp
    kullanılmayan/Drone.cpp
)

target_include_directories(${PROJECT_NAME} PRIVATE 
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/kullanılmayan
)

target_link_libraries(${PROJECT_NAME} PRIVATE raylib)

file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/resources DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
