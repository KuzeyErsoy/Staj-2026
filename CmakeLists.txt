cmake_minimum_required(VERSION 3.14)
project(Staj26)

set(CMAKE_CXX_STANDARD 17)

set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)

add_subdirectory(thirdparty/raylib)

add_executable(${PROJECT_NAME}
    main.cpp
    DroneSystem.cpp

    # Modular Fleet Simulation Subsystems
    fleet/Drone.cpp
    fleet/ClusterManager.cpp
    fleet/MissileSystem.cpp
    fleet/HangarBase.cpp
    fleet/FleetCamera.cpp
    fleet/FleetHUD.cpp

    # Legacy Game Code (Archived in oyun/)
    oyun/Player.cpp
    oyun/Map.cpp
    oyun/CameraController.cpp
    oyun/Collectible.cpp
    oyun/Radar.cpp
    oyun/ParticleSystem.cpp
    oyun/AudioManager.cpp
    oyun/Enemy.cpp
    oyun/SaveSystem.cpp
    oyun/Drone.cpp
)

target_include_directories(${PROJECT_NAME} PRIVATE 
    ${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/oyun
)

target_link_libraries(${PROJECT_NAME} PRIVATE raylib)

file(COPY ${CMAKE_CURRENT_SOURCE_DIR}/resources DESTINATION ${CMAKE_CURRENT_BINARY_DIR})
